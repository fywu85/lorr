cmake_minimum_required(VERSION 3.16)
project(random05_pipeline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PILOT_IPO "Enable portable interprocedural optimization" OFF)
if(PILOT_IPO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT PILOT_IPO_SUPPORTED OUTPUT PILOT_IPO_ERROR LANGUAGES CXX)
  if(NOT PILOT_IPO_SUPPORTED)
    message(FATAL_ERROR "PILOT IPO is unsupported: ${PILOT_IPO_ERROR}")
  endif()
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
find_package(Boost REQUIRED COMPONENTS program_options system filesystem log log_setup)
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED)
add_library(random05_engine src/engine.cpp src/operations.cpp src/diagnostics.cpp src/components.cpp src/replan.cpp src/rescore.cpp src/window.cpp src/shared_rankings.cpp src/joint_moves.cpp src/matching.cpp src/cycles.cpp)
target_include_directories(random05_engine PUBLIC src simulator/inc ${Boost_INCLUDE_DIRS})
target_link_libraries(random05_engine PUBLIC Threads::Threads OpenMP::OpenMP_CXX)
target_compile_options(random05_engine PRIVATE -Wall -Wextra -Wpedantic)
file(GLOB SIMULATOR_SOURCES simulator/src/*.cpp)
list(FILTER SIMULATOR_SOURCES EXCLUDE REGEX "Evaluation.cpp$")
add_executable(lifelong_random05 ${SIMULATOR_SOURCES} src/Entry.cpp)
target_compile_definitions(lifelong_random05 PRIVATE PYTHON=0 BOOST_LOG_DYN_LINK)
target_link_libraries(lifelong_random05 PRIVATE random05_engine ${Boost_LIBRARIES})
include(CTest)
if(BUILD_TESTING)
  add_executable(random05_tests tests/engine_tests.cpp simulator/src/States.cpp)
  target_link_libraries(random05_tests PRIVATE random05_engine)
  add_test(NAME random05_tests COMMAND random05_tests)
endif()

# Offline counterfactual evaluator: never used by the competition entry.
add_executable(random05_decision_probe tools/decision_probe.cpp simulator/src/States.cpp)
target_link_libraries(random05_decision_probe PRIVATE random05_engine)

# Read-only guided-cost classification for independently replayed traces.
add_executable(random05_goal_probe tools/goal_probe.cpp simulator/src/States.cpp)
target_link_libraries(random05_goal_probe PRIVATE random05_engine)
